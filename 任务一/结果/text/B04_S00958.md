## 正文
I am using yolov8's tracking feature for 2 streams (2 videos). I load 2 separate models and infer it to 2 streams in the main thread. However, as two models share the same memory, the ids of the objects of the two models increase together. For example, in stream 1 the id is 1, 2, 3, 4, then the next object detected in stream 2 is 5 and so on. How can I make the 2 models work independently? Thank you! Here is the example code: from ultralytics import YOLO import cv2 model1 = YOLO('yolov8s.pt') model2 = YOLO('yolov8s.pt') cap1 = cv2.VideoCapture("videos/1.mp4") cap2 = cv2.VideoCapture("videos/2.mp4") def draw_results(results, frame): body_boxes = results[0].boxes.xywh.cpu() if results[0].boxes.id is not None: boxes_output = [] track_ids = results[0].boxes.id.int().cpu().tolist() for box, id in zip(body_boxes, track_ids): x, y, w, h = box x, y, w, h = int(x), int(y), int(w), int(h) boxes_output.append([x-w//2, y-h//2, x+w//2, y+h//2]) for box, id in zip(boxes_output, track_ids): frame = cv2.rectangle(frame, (box[0], box[1]), (box[2], box[3]), (0, 255, 255), 2) frame = cv2.putText(frame, str(id), (box[0], box[1]-15), cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 0, 255), 1, cv2.LINE_AA) return frame while True: ret, frame1 = cap1.read() ret, frame2 = cap2.read() results1 = model1.track(frame1, persist=True, iou = 0.5, classes = [1, 2, 3, 5, 7], verbose = False) frame1 = draw_results(results1, frame1) results2 = model2.track(frame2, persist=True, iou = 0.5, classes = [1, 2, 3, 5, 7], verbose = False) frame2 = draw_results(results2, frame2) cv2.imshow("Camera 1", frame1) cv2.imshow("Camera 2", frame2) k = cv2.waitKey(1) if k == 32: cv2.destroyAllWindows() break I tried inference in 2 different threads and the results remained the same.
