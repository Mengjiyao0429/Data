# SerializeField in singleton MonoBehaviour class

[machine translation English]
 Problem
 SingletonMono would create its gameobject after the game start, but my singleton class StageManager need to get Camera using [SerializeField] .
 How could I do that?
 Or is there some better method to get Camera Object in a singleton MonoBehaviour?
I'd heard to avoid mounting singleton Monobehaviour Script to a gameobject by hand.
 My SingletonMono Base Class
 public abstract class SingletonMono<T>: MonoBehaviour where T: MonoBehaviour
{
 private static T _instance;
 protected static bool ApplicationIsQuitting;
 protected static bool IsGlobal;
 static SingletonMono()
 {
 ApplicationIsQuitting = false;
 IsGlobal = false;
 }
 public static T Instance
 {
 get
 {
 if (ApplicationIsQuitting)
 {
 Debug.LogWarning($"Application is quitting, you cant operate a singleton mono object: [{typeof(T)}]");
 return _instance;
 }
 _instance = FindObjectOfType<T>();
 if (FindObjectsOfType<T>().Length > 1)
 {
 Debug.LogError($"the singleton type [{typeof(T)}] has more than one instances");
 return _instance;
 }
 if (_instance is not null) return _instance;
 var obj = new GameObject(typeof(T).ToString());
 _instance = obj.AddComponent<T>();
 if(IsGlobal) DontDestroyOnLoad(obj);
 return _instance;
 }
 }
 public void OnApplicationQuit()
 {
 ApplicationIsQuitting = true;
 }
}
